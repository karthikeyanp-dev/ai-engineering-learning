def retry(max_attempts: int = 3):
    def decorator(func):
        async def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return await func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:

                    await asyncio.sleep(2 ** attempt)
        return wrapper
    return decorator

{`import numpy as np

# Vectors as embeddings (simplified)
embedding_cat = np.random.randn(768)  # Typical embedding dimension
embedding_dog = embedding_cat + np.random.randn(768) * 0.1  # Similar
embedding_car = np.random.randn(768)  # Different concept

# Cosine similarity — the foundation of semantic search
def cosine_similarity(a: np.ndarray, b: np.ndarray) -> float:
    return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))

print(f"cat vs dog: {cosine_similarity(embedding_cat, embedding_dog):.4f}")
print(f"cat vs car: {cosine_similarity(embedding_cat, embedding_car):.4f}")

# Softmax — converts logits to probabilities (used in attention)
def softmax(x: np.ndarray) -> np.ndarray:
    exp_x = np.exp(x - np.max(x))  # Numerical stability
    return exp_x / exp_x.sum()

logits = np.array([2.0, 1.0, 0.1, -1.0])
probs = softmax(logits)
print(f"Softmax probabilities: {probs}")  # Sums to 1.0

# Matrix multiplication — core operation in transformers
# Attention: Q @ K.T / sqrt(d_k) then softmax then @ V
d_k = 64
Q = np.random.randn(10, d_k)  # 10 tokens, 64 dim queries
K = np.random.randn(10, d_k)  # 10 tokens, 64 dim keys
V = np.random.randn(10, d_k)  # 10 tokens, 64 dim values

attention_scores = Q @ K.T / np.sqrt(d_k)
attention_weights = np.apply_along_axis(softmax, 1, attention_scores)
output = attention_weights @ V
print(f"Attention output shape: {output.shape}")  # (10, 64)
`}

## Transformer Architecture Fundamentals

The **Transformer** (Vaswani et al., 2017) is the architecture behind every modern LLM. The key innovation is the **self-attention mechanism** — it allows every token to "attend to" every other token in the sequence.

### How Self-Attention Works

1. Each token is projected into three vectors: **Query (Q)**, **Key (K)**, **Value (V)**
2. Attention scores = Q × K^T / √d_k (how much should token i attend to token j?)
3. Apply softmax to get attention weights (probabilities that sum to 1)
4. Output = attention weights × V (weighted combination of values)

<Callout type="tip" title="Intuition for Attention" client:load>
  Think of attention like a search engine: the Query is your search query, Keys are document titles, and Values are document contents. The dot product between Q and K tells you which documents are most relevant, and you retrieve a weighted mix of their contents.
</Callout>

### Multi-Head Attention

Instead of one attention operation, transformers use multiple "heads" in parallel — each head can learn different types of relationships (syntax, semantics, positional patterns). The outputs are concatenated and projected.

### The Full Transformer Block

```
Input → Layer Norm → Multi-Head Attention → Residual Add
      → Layer Norm → Feed-Forward Network → Residual Add → Output
```

Modern LLMs stack 32-96+ of these blocks, with billions of parameters in the feed-forward layers.

  As an AI engineer (not a researcher), you need conceptual understanding — not the ability to implement transformers from scratch. Focus on understanding WHY things work so you can debug and optimize at the application level.
</Callout>

{`import numpy as np

# A minimal neural network to solidify fundamentals
class SimpleNN:
    def __init__(self, input_size: int, hidden_size: int, output_size: int):
        # Xavier initialization
        self.W1 = np.random.randn(input_size, hidden_size) * np.sqrt(2.0 / input_size)
        self.b1 = np.zeros(hidden_size)
        self.W2 = np.random.randn(hidden_size, output_size) * np.sqrt(2.0 / hidden_size)
        self.b2 = np.zeros(output_size)
    
    def relu(self, x):
        return np.maximum(0, x)
    
    def forward(self, x):
        self.z1 = x @ self.W1 + self.b1
        self.a1 = self.relu(self.z1)
        self.z2 = self.a1 @ self.W2 + self.b2
        return self.z2
    
    def predict(self, x):
        logits = self.forward(x)
        return np.argmax(logits, axis=1)

# Demo: XOR problem (non-linearly separable)
X = np.array([[0,0], [0,1], [1,0], [1,1]])
y = np.array([0, 1, 1, 0])

nn = SimpleNN(input_size=2, hidden_size=8, output_size=2)
print(f"Random predictions: {nn.predict(X)}")
print("(Training omitted — focus is on architecture understanding)")
